\begin{frame}{Organização}
    \begin{itemize}
        \item As apresentações no Beamer são organizadas em \texttt{frame}s;
        \item Cada \texttt{frame} contém pelo menos um slide;
        \item Se um \texttt{frame} tiver mais conteúdo do que cabe num slide, ou se for usado o recurso de animação (ex.: itens aparecendo na tela um de cada vez), o \texttt{frame} vai ser partido em mais slides.
    \end{itemize}
\end{frame}

\begin{frame}{Algoritmos em pseudocódigo}
    \begin{columns}
        \begin{column}{0.4\textwidth}
            O Algoritmo~\ref{alg:fft} mostra um exemplo de pseudocódigo usando o pacote \texttt{algorithm2e}.
        \end{column}
        \begin{column}{0.6\textwidth}
            \begin{algorithm}[H]
                \caption{FFT de Cooley-Tukey}
                \label{alg:fft}
                \DontPrintSemicolon
                \SetKwProg{Fn}{function}{:}{}
                \SetKwFunction{FFT}{fft}
                \KwIn{$x[0 ... n-1]$, tal que  $x[i] \in \mathbb{C}, n = 2^c, c \in \mathbb{N}$}
                \KwOut{$y[0...n-1]$, tal que  $y[i] \in \mathbb{C}$}
                \Fn{\FFT{$x$}}{
                    \If{$n = 1$}{
                        \KwRet $x$\;
                    }
                    $y_e \gets \FFT{x[0,2,...,n-2]}$\;
                    $y_o \gets \FFT{x[1,3,...,n-1]}$\;
                    \For{$k \gets 0$ to $\frac{n}{2} - 1$}{
                        $w \gets e^{\frac{2 \pi k i}{n}}$\;
                        $y[k] \gets y_e[k] + y_o[k] \cdot w$\;
                        $y[k+\frac{n}{2}] \gets y_e[k] - y_o[k] \cdot w$\;
                    }
                    \KwRet $y$\;
                }
            \end{algorithm}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[fragile]{Código}
    \begin{columns}
        \begin{column}{0.35\textwidth}
            Já a Listagem~\ref{lst:fft} mostra o mesmo código do slide anterior, porém em C++, usando um bloco de código formatado com o pacote \texttt{listings}.

            \begin{alertblock}{\scriptsize Importante}
                \scriptsize Veja que neste \texttt{frame} foi necessário habilitar a opção \verb|[fragile]|.
            \end{alertblock}
        \end{column}
        \begin{column}{0.65\textwidth}
            \begin{lstlisting}[language=C++, caption=\lstcaption{FFT de Cooley-Tukey em C++}, label=lst:fft]
void fft(Complex x[], int n, int &op) {
  if (n <= 1)
    return;

  Complex x_e[n/2], x_o[n/2];
  for (int i = 0; i < n/2; ++i)
    x_e[i] = x[i*2], x_o[i] = x[i*2 + 1];
  fft(x_e, n/2, op);
  fft(x_o, n/2, op);

  for (int k = 0; k < n/2; ++k) {
    ++op;
    Complex w = exp(Complex(0, -2*M_PI*k / n));
    x[k] = x_e[k] + x_o[k] * w;
    x[k + n/2] = x_e[k] - x_o[k] * w;
  }
}\end{lstlisting}
        \end{column}
    \end{columns}
\end{frame}

    \end{block}
